# 퀴즈 목록

`cli` 모듈에서 퀴즈 목록 보여주는 함수이자 추가 상호작용을 `quizgame`모듈을 통해 구현했습니다.

``` python
def list_menu(manager: QuizGameManager):
    '''퀴즈 목록'''
    print("\n=== 퀴즈 목록 ===")
    for i, quiz in enumerate(manager.quizzes, start=1):
        print(f"{quiz.to_string(i, True, True)}\n")
    print("1. 퀴즈 추가")
    print("2. 퀴즈 삭제")
    print("3. 메인 메뉴")
    try:
        choice = int(input("선택: "))
    except ValueError:
        print("다시 입력해주세요.")
        return list_menu(manager)
    except (KeyboardInterrupt, EOFError):
        manager.save()
        print("게임을 종료합니다.\n")
              
    if choice == 1:
        add_menu(manager)
    elif choice == 2:
        remove_menu(manager)
    elif choice == 3:
        return main_menu(manager)
    else:
        print("다시 입력해주세요.")
        return list_menu(manager)
```
